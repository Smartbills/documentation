## Installation

```bash
npm install @smartbills/react-native-sdk @smartbills/sdk
```

### Additional Dependencies

```bash
npm install react-native-camera react-native-image-picker
```

For iOS, run:

```bash
cd ios && pod install
```

## Quick Start

### Provider Setup

```tsx
import { SmartbillsReactNativeProvider } from '@smartbills/react-native-sdk';

function App() {
  return (
    <SmartbillsReactNativeProvider
      apiKey="sk_live_xxxxx"
      environment="production"
    >
      <Navigation />
    </SmartbillsReactNativeProvider>
  );
}
```

## Camera Integration

### Receipt Scanner

```tsx
import { ReceiptScanner } from '@smartbills/react-native-sdk';

function ScanScreen() {
  const handleCapture = async (expense) => {
    console.log('Expense created:', expense);
    navigation.navigate('ExpenseDetail', { id: expense.id });
  };

  return (
    <ReceiptScanner
      businessId={123}
      onCapture={handleCapture}
      onError={(error) => Alert.alert('Error', error.message)}
      autoProcess={true}
      showGuide={true}
    />
  );
}
```

### Manual Camera Control

```tsx
import { useReceiptCamera } from '@smartbills/react-native-sdk';

function CustomScanScreen() {
  const {
    capture,
    isProcessing,
    cameraRef,
    focusPoint,
    setFocusPoint
  } = useReceiptCamera();

  const handleCapture = async () => {
    const expense = await capture({
      businessId: 123,
      autoProcess: true
    });
    console.log('Created expense:', expense);
  };

  return (
    <View style={styles.container}>
      <Camera
        ref={cameraRef}
        style={styles.camera}
        onTap={setFocusPoint}
      />
      {focusPoint && <FocusIndicator point={focusPoint} />}
      <CaptureButton
        onPress={handleCapture}
        loading={isProcessing}
      />
    </View>
  );
}
```

## Offline Mode

The React Native SDK supports offline-first architecture:

```tsx
import { useOfflineExpenses } from '@smartbills/react-native-sdk';

function OfflineExpenseList() {
  const {
    expenses,
    pendingUploads,
    syncStatus,
    sync
  } = useOfflineExpenses();

  return (
    <View>
      <Text>Sync Status: {syncStatus}</Text>
      <Text>Pending Uploads: {pendingUploads.length}</Text>
      
      <Button title="Sync Now" onPress={sync} />
      
      <FlatList
        data={expenses}
        renderItem={({ item }) => <ExpenseCard expense={item} />}
      />
    </View>
  );
}
```

### Offline Configuration

```tsx
<SmartbillsReactNativeProvider
  apiKey="sk_live_xxxxx"
  offlineConfig={{
    enabled: true,
    syncInterval: 30000,
    maxPendingUploads: 50,
    onSyncComplete: (results) => console.log('Synced:', results)
  }}
>
  <App />
</SmartbillsReactNativeProvider>
```

## Hooks

### useExpenses

```tsx
import { useExpenses } from '@smartbills/react-native-sdk';

function ExpenseListScreen() {
  const {
    expenses,
    isLoading,
    error,
    refetch,
    loadMore,
    hasMore
  } = useExpenses({ businessId: 123 });

  return (
    <FlatList
      data={expenses}
      renderItem={({ item }) => <ExpenseCard expense={item} />}
      onRefresh={refetch}
      refreshing={isLoading}
      onEndReached={hasMore ? loadMore : undefined}
    />
  );
}
```

### useExpenseReports

```tsx
import { useExpenseReports } from '@smartbills/react-native-sdk';

function ReportsScreen() {
  const { reports, submitReport, isSubmitting } = useExpenseReports();

  const handleSubmit = async (reportId) => {
    await submitReport(reportId);
    Alert.alert('Success', 'Report submitted!');
  };

  return (
    <FlatList
      data={reports}
      renderItem={({ item }) => (
        <ReportCard
          report={item}
          onSubmit={() => handleSubmit(item.id)}
          submitting={isSubmitting}
        />
      )}
    />
  );
}
```

## Push Notifications

```tsx
import { useExpenseNotifications } from '@smartbills/react-native-sdk';

function NotificationHandler() {
  useExpenseNotifications({
    onReportApproved: (report) => {
      Alert.alert('Report Approved', `${report.title} was approved!`);
    },
    onReportRejected: (report) => {
      Alert.alert('Report Rejected', report.rejectionReason);
    },
    onExpenseProcessed: (expense) => {
      console.log('Expense processed:', expense);
    }
  });

  return null;
}
```

## Permissions

The SDK handles camera and storage permissions automatically, but you can customize:

```tsx
import { usePermissions } from '@smartbills/react-native-sdk';

function PermissionGate({ children }) {
  const { 
    camera, 
    storage,
    requestCamera,
    requestStorage
  } = usePermissions();

  if (!camera.granted) {
    return (
      <View>
        <Text>Camera permission required</Text>
        <Button title="Grant Access" onPress={requestCamera} />
      </View>
    );
  }

  return children;
}
```

## Best Practices

<Tip>
**Enable offline mode** - Users often capture receipts without connectivity.
</Tip>

<Tip>
**Handle permissions gracefully** - Always provide fallback options.
</Tip>

<Tip>
**Optimize images** - Use compression before uploading large receipts.
</Tip>

<Tip>
**Test on real devices** - Camera functionality differs from simulators.
</Tip>

## Platform-Specific Configuration

### iOS (Info.plist)

```xml
<key>NSCameraUsageDescription</key>
<string>We need camera access to scan receipts</string>
<key>NSPhotoLibraryUsageDescription</key>
<string>We need photo library access to select receipt images</string>
```

### Android (AndroidManifest.xml)

```xml
<uses-permission android:name="android.permission.CAMERA" />
<uses-permission android:name="android.permission.READ_EXTERNAL_STORAGE" />
<uses-permission android:name="android.permission.WRITE_EXTERNAL_STORAGE" />
```
