---
title: "Créer une facture"
api: "POST https://api.smartbills.io/v1/invoices"
description: "Créer une nouvelle facture"
---

## Paramètres du corps

<ParamField body="invoiceNumber" type="string">
  Numéro de facture personnalisé. Si non fourni, un numéro sera généré automatiquement.
</ParamField>

<ParamField body="customerId" type="integer">
  ID du client à facturer
</ParamField>

<ParamField body="date" type="string">
  Date d'émission de la facture au format ISO 8601. Par défaut, la date actuelle.
</ParamField>

<ParamField body="dueDate" type="string">
  Date d'échéance du paiement au format ISO 8601. Calculée automatiquement à partir de `paymentTerms` si non fournie.
</ParamField>

<ParamField body="paymentTerms" type="string">
  Conditions de paiement de la facture. Lorsque défini, `dueDate` est calculée automatiquement à partir de la date d'émission si non explicitement fournie. Valeurs : `UponReceipt`, `Net7`, `Net15`, `Net30`, `Net45`, `Net60`, `Net90`, `Custom`
</ParamField>

<ParamField body="currency" type="string">
  Code de devise ISO 4217 (ex. `CAD`, `USD`). Par défaut, la devise de l'entreprise.
</ParamField>

<ParamField body="note" type="string">
  Mémo ou notes à inclure sur la facture
</ParamField>

<ParamField body="billingPeriodStartDate" type="string">
  Début de la période de facturation (ISO 8601)
</ParamField>

<ParamField body="billingPeriodEndDate" type="string">
  Fin de la période de facturation (ISO 8601)
</ParamField>

<ParamField body="autoPaymentDate" type="string">
  Date de prélèvement automatique du paiement (ISO 8601)
</ParamField>

<ParamField body="amountDue" type="object">
  <Expandable title="propriétés">
    <ParamField body="amount" type="number" required>Montant dû</ParamField>
    <ParamField body="currency" type="string" required>Code de devise ISO 4217</ParamField>
  </Expandable>
</ParamField>

<ParamField body="balanceForward" type="object">
  <Expandable title="propriétés">
    <ParamField body="amount" type="number">Solde reporté</ParamField>
    <ParamField body="currency" type="string">Code de devise</ParamField>
  </Expandable>
</ParamField>

<ParamField body="paymentLinkUrl" type="string">
  URL de lien de paiement personnalisé
</ParamField>

<ParamField body="lineItems" type="array">
  <Expandable title="propriétés">
    <ParamField body="description" type="string" required>Description de la ligne d'article</ParamField>
    <ParamField body="quantity" type="number" required>Quantité</ParamField>
    <ParamField body="unitPrice" type="number" required>Prix unitaire en centimes</ParamField>
    <ParamField body="productId" type="integer">ID du produit associé</ParamField>
    <ParamField body="taxable" type="boolean">Si la ligne est taxable</ParamField>
    <ParamField body="taxIds" type="array">Tableau d'identifiants de taxes à appliquer</ParamField>
  </Expandable>
</ParamField>

<ParamField body="fees" type="array">
  <Expandable title="propriétés">
    <ParamField body="feeId" type="integer">ID d'un frais du catalogue de l'entreprise</ParamField>
    <ParamField body="name" type="string">Nom du frais (requis si aucun feeId)</ParamField>
    <ParamField body="type" type="string">Type de frais : `FIXED` ou `PERCENTAGE`</ParamField>
    <ParamField body="amount" type="number">Montant fixe du frais</ParamField>
    <ParamField body="percentage" type="number">Pourcentage du frais (quand le type est `PERCENTAGE`)</ParamField>
  </Expandable>
</ParamField>

<ParamField body="discounts" type="array">
  <Expandable title="propriétés">
    <ParamField body="promoCodeId" type="integer">ID d'un code promo à appliquer</ParamField>
    <ParamField body="name" type="string">Nom de la remise (requis si aucun promoCodeId)</ParamField>
    <ParamField body="type" type="string">Type de remise : `FIXED` ou `PERCENTAGE`</ParamField>
    <ParamField body="amount" type="number">Montant fixe de la remise</ParamField>
    <ParamField body="percentage" type="number">Pourcentage de la remise (quand le type est `PERCENTAGE`)</ParamField>
  </Expandable>
</ParamField>

<ParamField body="reminders" type="array">
  <Expandable title="propriétés">
    <ParamField body="enabled" type="boolean" required>Si le rappel est actif</ParamField>
    <ParamField body="timing" type="string" required>Quand envoyer : `BEFORE_DUE`, `ON_DUE`, `AFTER_DUE`</ParamField>
    <ParamField body="daysOffset" type="integer" required>Nombre de jours avant/après la date d'échéance</ParamField>
    <ParamField body="frequency" type="string" required>Fréquence du rappel : `ONCE`, `WEEKLY`, `BIWEEKLY`, `MONTHLY`</ParamField>
    <ParamField body="maxReminders" type="integer" required>Nombre maximum de courriels de rappel</ParamField>
    <ParamField body="message" type="string">Message de rappel personnalisé</ParamField>
  </Expandable>
</ParamField>

## Réponse

Retourne l'objet facture créé.

<RequestExample>

```bash cURL
curl --request POST \
  --url https://api.smartbills.io/v1/invoices \
  --header 'Authorization: Bearer YOUR_API_KEY' \
  --header 'Content-Type: application/json' \
  --data '{
    "customerId": 789,
    "currency": "CAD",
    "paymentTerms": "Net30",
    "date": "2026-04-01T00:00:00Z",
    "lineItems": [
      {
        "description": "Services de développement web",
        "quantity": 10,
        "unitPrice": 15000,
        "taxable": true,
        "taxIds": [1]
      }
    ],
    "fees": [
      { "name": "Frais de service", "type": "PERCENTAGE", "percentage": 5 }
    ],
    "discounts": [
      { "name": "Remise fidélité", "type": "FIXED", "amount": 500 }
    ]
  }'
```

```javascript JavaScript
const invoice = await smartbills.invoices.create({
  customerId: 789,
  currency: "CAD",
  paymentTerms: "Net30",
  date: "2026-04-01T00:00:00Z",
  lineItems: [
    { description: "Services de développement web", quantity: 10, unitPrice: 15000, taxable: true, taxIds: [1] }
  ],
  fees: [
    { name: "Frais de service", type: "PERCENTAGE", percentage: 5 }
  ],
  discounts: [
    { name: "Remise fidélité", type: "FIXED", amount: 500 }
  ]
});
```
