---
title: "Lister les approbations en attente"
description: "Récupérer une liste de rapports de dépenses en attente de votre approbation"
api: "GET https://api.smartbills.io/v1/businesses/{businessId}/approbations/pending"
---

## Aperçu

Récupérer tous les rapports de dépenses en attente de votre approbation. Ce point de terminaison retourné les rapports soumis par les employés qui font partie de votre chaîne d'approbation. Les résultats sont triés par date de soumission, les plus récents en premier.

## Authentification

<ParamField header="Authorization" type="string" required>
  Jeton Bearer pour l'authentification. Format : `Bearer YOUR_API_KEY`
</ParamField>

## Paramètres de chemin

<ParamField path="businessId" type="integer" required>
  L'identifiant unique de l'entreprise
</ParamField>

## Paramètres de requête

<ParamField query="page" type="integer" default="1">
  Numéro de page pour la pagination
</ParamField>

<ParamField query="pageSize" type="integer" default="20">
  Nombre d'éléments par page (max : 100)
</ParamField>

<ParamField query="employéeId" type="integer">
  Filtrer par identifiant de l'employé soumetteur
</ParamField>

<ParamField query="minAmount" type="number">
  Filtrer par montant minimum du rapport
</ParamField>

<ParamField query="maxAmount" type="number">
  Filtrer par montant maximum du rapport
</ParamField>

<ParamField query="sortBy" type="string" default="submittedAt">
  Champ de tri. Options : `submittedAt`, `totalAmount`, `employéeName`
</ParamField>

<ParamField query="sortOrder" type="string" default="desc">
  Ordre de tri. Options : `asc`, `desc`
</ParamField>

## Réponse

<ResponseField name="data" type="array">
  Tableau d'objets d'approbation en attente

  <Expandable title="Objet Approbation en attente">
    <ResponseField name="reportId" type="integer">
      Identifiant du rapport de dépenses
    </ResponseField>

    <ResponseField name="reportName" type="string">
      Nom du rapport de dépenses
    </ResponseField>

    <ResponseField name="employéeId" type="integer">
      Identifiant de l'employé qui à soumis le rapport
    </ResponseField>

    <ResponseField name="employéeName" type="string">
      Nom complet de l'employé soumetteur
    </ResponseField>

    <ResponseField name="totalAmount" type="number">
      Montant total du rapport
    </ResponseField>

    <ResponseField name="currency" type="string">
      Code de devise ISO 4217
    </ResponseField>

    <ResponseField name="expenseCount" type="integer">
      Nombre de dépenses dans le rapport
    </ResponseField>

    <ResponseField name="submittedAt" type="string">
      Horodatage de soumission du rapport (ISO 8601)
    </ResponseField>

    <ResponseField name="description" type="string">
      Description du rapport
    </ResponseField>

    <ResponseField name="approvalLevel" type="integer">
      Niveau d'approbation actuel dans la chaîne (1 = premier approbateur, 2 = deuxième, etc.)
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="metadata" type="object">
  Métadonnées de pagination

  <Expandable title="Objet Métadonnées">
    <ResponseField name="page" type="integer">
      Numéro de page actuel
    </ResponseField>
    <ResponseField name="pageSize" type="integer">
      Éléments par page
    </ResponseField>
    <ResponseField name="totalPages" type="integer">
      Nombre total de pages
    </ResponseField>
    <ResponseField name="count" type="integer">
      Nombre total d'approbations en attente
    </ResponseField>
  </Expandable>
</ResponseField>
