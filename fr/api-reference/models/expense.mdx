## Dépense

L'objet Dépense représente une dépense ou transaction professionnelle unique. Les dépenses peuvent être autonomes ou regroupées dans des rapports de dépenses pour approbation et remboursement.

## Propriétés

<ResponseField name="id" type="integer">
  Identifiant unique de la dépense
</ResponseField>

<ResponseField name="businessId" type="integer">
  ID de l'entreprise à laquelle cette dépense appartient
</ResponseField>

<ResponseField name="userId" type="integer">
  ID de l'utilisateur qui a créé la dépense
</ResponseField>

<ResponseField name="merchant" type="string">
  Nom du marchand ou du fournisseur où la dépense a été engagée
</ResponseField>

<ResponseField name="amount" type="number">
  Montant total de la dépense
</ResponseField>

<ResponseField name="currency" type="string">
  Code de devise ISO 4217 à trois lettres (p. ex., `USD`, `CAD`, `EUR`)
</ResponseField>

<ResponseField name="date" type="string">
  Date de la transaction au format ISO 8601 (`AAAA-MM-JJ`)
</ResponseField>

<ResponseField name="category" type="object">
  Affectation de la catégorie de dépense
  <Expandable title="Objet Catégorie">
    <ResponseField name="id" type="integer">
      ID de la catégorie
    </ResponseField>
    <ResponseField name="name" type="string">
      Nom de la catégorie
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="status" type="string">
  Statut actuel de la dépense. Valeurs possibles : `pending`, `submitted`, `approved`, `rejected`, `reimbursed`
</ResponseField>

<ResponseField name="paymentMethod" type="string">
  Mode de paiement de la dépense. Valeurs possibles : `cash`, `credit_card`, `debit_card`, `bank_transfer`, `personal_card`
</ResponseField>

<ResponseField name="notes" type="string">
  Notes ou description supplémentaires pour la dépense
</ResponseField>

<ResponseField name="tags" type="array">
  Tableau d'étiquettes pour la catégorisation et le filtrage
</ResponseField>

<ResponseField name="reportId" type="integer">
  ID du rapport de dépenses auquel cette dépense est assignée. `null` si non assignée.
</ResponseField>

<ResponseField name="attachments" type="array">
  Tableau de pièces jointes de reçus ou de documents
  <Expandable title="Objet Pièce jointe">
    <ResponseField name="id" type="integer">
      ID de la pièce jointe
    </ResponseField>
    <ResponseField name="filename" type="string">
      Nom de fichier original
    </ResponseField>
    <ResponseField name="url" type="string">
      URL de téléchargement de la pièce jointe
    </ResponseField>
    <ResponseField name="thumbnailUrl" type="string">
      URL de la miniature de la pièce jointe
    </ResponseField>
    <ResponseField name="contentType" type="string">
      Type MIME du fichier
    </ResponseField>
    <ResponseField name="size" type="integer">
      Taille du fichier en octets
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="tax" type="object">
  Informations fiscales de la dépense
  <Expandable title="Objet Taxe">
    <ResponseField name="amount" type="number">
      Montant de la taxe
    </ResponseField>
    <ResponseField name="rate" type="number">
      Taux de taxe en pourcentage
    </ResponseField>
    <ResponseField name="name" type="string">
      Nom de la taxe (p. ex., "TPS", "TVA", "Taxe de vente")
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="vendorId" type="integer">
  ID de l'enregistrement fournisseur lié, le cas échéant
</ResponseField>

<ResponseField name="department" type="string">
  Département auquel la dépense est attribuée
</ResponseField>

<ResponseField name="createdAt" type="string">
  Horodatage de création de la dépense (ISO 8601)
</ResponseField>

<ResponseField name="updatedAt" type="string">
  Horodatage de la dernière mise à jour de la dépense (ISO 8601)
</ResponseField>
