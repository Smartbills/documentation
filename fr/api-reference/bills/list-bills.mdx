---
title: "Lister les factures"
description: "Récupérer une liste paginées de factures"
api: "GET https://api.smartbills.io/v1/bills"
---

## Aperçu

Récupérer une liste paginées de factures pour l'utilisateur authentifié. Vous pouvez filtrer par fournisseur, statut, plage de dates d'échéance et plage de montants pour trouver des factures spécifiques.

## Authentification

<ParamField header="Authorization" type="string" required>
  Jeton Bearer pour l'authentification. Format : `Bearer YOUR_API_KEY`
</ParamField>

## Paramètres de requête

<ParamField query="page" type="integer" default="1">
  Numéro de page pour la pagination
</ParamField>

<ParamField query="pageSize" type="integer" default="20">
  Nombre d'éléments par page (max : 100)
</ParamField>

<ParamField query="businessId" type="integer">
  Filtrer les factures par identifiant d'entreprise
</ParamField>

<ParamField query="vendorId" type="integer">
  Filtrer par identifiant de fournisseur
</ParamField>

<ParamField query="status" type="string">
  Filtrer par statut de facture. Options : `draft`, `pending`, `paid`, `overdue`, `cancelled`
</ParamField>

<ParamField query="dueDateFrom" type="string">
  Filtrer les factures dues à partir de cette date (AAAA-MM-JJ)
</ParamField>

<ParamField query="dueDateTo" type="string">
  Filtrer les factures dues jusqu'à cette date (AAAA-MM-JJ)
</ParamField>

<ParamField query="minAmount" type="number">
  Montant minimum de la facture (inclus)
</ParamField>

<ParamField query="maxAmount" type="number">
  Montant maximum de la facture (inclus)
</ParamField>

<ParamField query="search" type="string">
  Rechercher dans le nom du fournisseur, le numéro de facture et les notes
</ParamField>

<ParamField query="sortBy" type="string" default="dueDate">
  Champ de tri. Options : `dueDate`, `amount`, `createdAt`, `updatedAt`
</ParamField>

<ParamField query="sortOrder" type="string" default="desc">
  Ordre de tri. Options : `asc`, `desc`
</ParamField>

    <ResponseField name="businessId" type="integer">

        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="amount" type="number">
      Montant total de la facture
    </ResponseField>

    <ResponseField name="currency" type="string">
      Code de devise ISO 4217

    <ResponseField name="dueDate" type="string">
      Date d'échéance du paiement (AAAA-MM-JJ)

    <ResponseField name="status" type="string">

    </ResponseField>

    <ResponseField name="createdAt" type="string">
      Horodatage de la création de la facture (ISO 8601)

    <ResponseField name="updatedAt" type="string">

    <ResponseField name="totalPages" type="integer">
      Nombre total de pages
    </ResponseField>
    <ResponseField name="count" type="integer">
      Nombre total de factures correspondant aux filtres
    </ResponseField>
  </Expandable>
</ResponseField>

