---
title: "List Bills"
description: "Retrieve a paginated list of bills"
api: "GET https://api.smartbills.io/v1/bills"
---

## Overview

Retrieve a paginated list of bills for the authenticated user. You can filter by vendor, status, due date range, and amount range to find specific bills.

## Authentication

<ParamField header="Authorization" type="string" required>
  Bearer token for authentication. Format: `Bearer YOUR_API_KEY`
</ParamField>

## Query Parameters

<ParamField query="page" type="integer" default="1">
  Page number for pagination
</ParamField>

<ParamField query="pageSize" type="integer" default="20">
  Number of items per page (max: 100)
</ParamField>

<ParamField query="businessId" type="integer">
  Filter bills by business ID
</ParamField>

<ParamField query="vendorId" type="integer">
  Filter by vendor ID
</ParamField>

<ParamField query="status" type="string">
  Filter by bill status. Options: `draft`, `pending`, `paid`, `overdue`, `cancelled`
</ParamField>

<ParamField query="dueDateFrom" type="string">
  Filter bills due from this date (YYYY-MM-DD)
</ParamField>

<ParamField query="dueDateTo" type="string">
  Filter bills due until this date (YYYY-MM-DD)
</ParamField>

<ParamField query="minAmount" type="number">
  Minimum bill amount (inclusive)
</ParamField>

<ParamField query="maxAmount" type="number">
  Maximum bill amount (inclusive)
</ParamField>

<ParamField query="search" type="string">
  Search across vendor name, bill number, and notes
</ParamField>

<ParamField query="sortBy" type="string" default="dueDate">
  Field to sort by. Options: `dueDate`, `amount`, `createdAt`, `updatedAt`
</ParamField>

<ParamField query="sortOrder" type="string" default="desc">
  Sort order. Options: `asc`, `desc`
</ParamField>

```bash cURL
  --header 'Authorization: Bearer YOUR_API_KEY'
```
