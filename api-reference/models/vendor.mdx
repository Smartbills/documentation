## Vendor

The Vendor object represents an external supplier or service provider that your business transacts with. Vendors can be linked to expenses and bills for tracking and reporting.

## Properties

<ResponseField name="id" type="integer">
  Unique identifier for the vendor
</ResponseField>

<ResponseField name="businessId" type="integer">
  ID of the business this vendor belongs to
</ResponseField>

<ResponseField name="name" type="string">
  Vendor's display name
</ResponseField>

<ResponseField name="address" type="object">
  Vendor's mailing address
  <Expandable title="Address Object">
    <ResponseField name="street" type="string">
      Street address
    </ResponseField>
    <ResponseField name="city" type="string">
      City
    </ResponseField>
    <ResponseField name="state" type="string">
      State or province
    </ResponseField>
    <ResponseField name="postalCode" type="string">
      Postal or ZIP code
    </ResponseField>
    <ResponseField name="country" type="string">
      Two-letter ISO 3166 country code
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="taxId" type="string">
  Vendor's tax identification number (EIN, VAT number, etc.)
</ResponseField>

<ResponseField name="email" type="string">
  Primary contact email address
</ResponseField>

<ResponseField name="phone" type="string">
  Contact phone number
</ResponseField>

<ResponseField name="website" type="string">
  Vendor's website URL
</ResponseField>

<ResponseField name="defaultCategoryId" type="integer">
  Default expense category ID automatically applied to expenses from this vendor
</ResponseField>

<ResponseField name="logo" type="string">
  URL of the vendor's logo image
</ResponseField>

<ResponseField name="createdAt" type="string">
  Timestamp when the vendor was created (ISO 8601)
</ResponseField>

<ResponseField name="updatedAt" type="string">
  Timestamp when the vendor was last updated (ISO 8601)
</ResponseField>
