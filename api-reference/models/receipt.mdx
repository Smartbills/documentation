## Receipt

The Receipt object represents a digitized receipt that has been processed through OCR. Receipts contain extracted transaction data and are linked to expenses for documentation and audit purposes.

## Properties

<ResponseField name="id" type="integer">
  Unique identifier for the receipt
</ResponseField>

<ResponseField name="businessId" type="integer">
  ID of the business this receipt belongs to
</ResponseField>

<ResponseField name="merchant" type="string">
  Merchant name extracted from the receipt
</ResponseField>

<ResponseField name="amount" type="number">
  Total amount extracted from the receipt
</ResponseField>

<ResponseField name="currency" type="string">
  Currency code detected from the receipt (ISO 4217)
</ResponseField>

<ResponseField name="date" type="string">
  Transaction date extracted from the receipt (`YYYY-MM-DD`)
</ResponseField>

<ResponseField name="taxes" type="array">
  Array of tax entries detected on the receipt
  <Expandable title="Tax Entry">
    <ResponseField name="name" type="string">
      Tax name (e.g., "GST", "QST", "VAT", "Sales Tax")
    </ResponseField>
    <ResponseField name="amount" type="number">
      Tax amount
    </ResponseField>
    <ResponseField name="rate" type="number">
      Tax rate as a percentage
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="items" type="array">
  Individual line items extracted from the receipt
  <Expandable title="Line Item">
    <ResponseField name="description" type="string">
      Item description
    </ResponseField>
    <ResponseField name="quantity" type="number">
      Quantity purchased
    </ResponseField>
    <ResponseField name="unitPrice" type="number">
      Price per unit
    </ResponseField>
    <ResponseField name="total" type="number">
      Total for this line item
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="imageUrl" type="string">
  URL of the stored receipt image
</ResponseField>

<ResponseField name="ocrData" type="object">
  Raw OCR processing metadata
  <Expandable title="OCR Data">
    <ResponseField name="confidence" type="number">
      Overall OCR confidence score (0.0 to 1.0)
    </ResponseField>
    <ResponseField name="rawText" type="string">
      Full raw text extracted from the receipt
    </ResponseField>
    <ResponseField name="processingTime" type="integer">
      OCR processing time in milliseconds
    </ResponseField>
    <ResponseField name="engine" type="string">
      OCR engine used for processing
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="createdAt" type="string">
  Timestamp when the receipt was processed (ISO 8601)
</ResponseField>
