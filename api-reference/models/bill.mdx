## Bill

The Bill object represents a vendor invoice or bill that your business needs to pay. Bills track amounts owed, due dates, line items, and payment status.

## Properties

<ResponseField name="id" type="integer">
  Unique identifier for the bill
</ResponseField>

<ResponseField name="businessId" type="integer">
  ID of the business this bill belongs to
</ResponseField>

<ResponseField name="vendorId" type="integer">
  ID of the vendor who issued the bill
</ResponseField>

<ResponseField name="vendor" type="object">
  Vendor summary information
  <Expandable title="Vendor Summary">
    <ResponseField name="id" type="integer">
      Vendor ID
    </ResponseField>
    <ResponseField name="name" type="string">
      Vendor display name
    </ResponseField>
    <ResponseField name="email" type="string">
      Vendor contact email
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="billNumber" type="string">
  Vendor's invoice or bill reference number
</ResponseField>

<ResponseField name="amount" type="number">
  Total bill amount including tax
</ResponseField>

<ResponseField name="currency" type="string">
  ISO 4217 currency code (e.g., `USD`, `CAD`, `EUR`)
</ResponseField>

<ResponseField name="dueDate" type="string">
  Payment due date (`YYYY-MM-DD`)
</ResponseField>

<ResponseField name="issueDate" type="string">
  Date the bill was issued by the vendor (`YYYY-MM-DD`)
</ResponseField>

<ResponseField name="status" type="string">
  Current bill status. Possible values: `draft`, `pending`, `paid`, `overdue`, `cancelled`
</ResponseField>

<ResponseField name="notes" type="string">
  Notes or description for the bill
</ResponseField>

<ResponseField name="lineItems" type="array">
  Individual line items on the bill
  <Expandable title="Line Item Object">
    <ResponseField name="id" type="integer">
      Line item ID
    </ResponseField>
    <ResponseField name="description" type="string">
      Item description
    </ResponseField>
    <ResponseField name="quantity" type="number">
      Quantity ordered
    </ResponseField>
    <ResponseField name="unitPrice" type="number">
      Price per unit
    </ResponseField>
    <ResponseField name="amount" type="number">
      Total for this line item (quantity x unitPrice)
    </ResponseField>
    <ResponseField name="categoryId" type="integer">
      Category ID for this line item
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="attachments" type="array">
  Attached files (original invoice, supporting documents)
  <Expandable title="Attachment Object">
    <ResponseField name="id" type="integer">
      Attachment ID
    </ResponseField>
    <ResponseField name="filename" type="string">
      Original filename
    </ResponseField>
    <ResponseField name="url" type="string">
      Download URL
    </ResponseField>
    <ResponseField name="contentType" type="string">
      MIME type
    </ResponseField>
    <ResponseField name="size" type="integer">
      File size in bytes
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="subtotal" type="number">
  Subtotal before tax
</ResponseField>

<ResponseField name="taxAmount" type="number">
  Total tax amount
</ResponseField>

<ResponseField name="paidAt" type="string">
  Timestamp when the bill was paid (ISO 8601). `null` if unpaid.
</ResponseField>

<ResponseField name="createdAt" type="string">
  Timestamp when the bill was created (ISO 8601)
</ResponseField>

<ResponseField name="updatedAt" type="string">
  Timestamp when the bill was last updated (ISO 8601)
</ResponseField>

## Status Lifecycle

```
draft -> pending -> paid
                 -> overdue -> paid
                            -> cancelled
      -> cancelled
```
