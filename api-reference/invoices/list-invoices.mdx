---
title: "List Invoices"
api: "GET https://api.smartbills.io/v1/businesses/{businessId}/invoices"
description: "Retrieve a paginated list of invoices for a business"
---

## Path Parameters

<ParamField path="businessId" type="integer" required>
  The unique identifier of the business
</ParamField>

## Query Parameters

<ParamField query="page" type="integer" default="1">
  Page number for pagination
</ParamField>

<ParamField query="pageSize" type="integer" default="25">
  Number of items per page (max 100)
</ParamField>

<ParamField query="status" type="string">
  Filter by invoice status: `Draft`, `Sent`, `Paid`, `Overdue`, `Void`, `Cancelled`
</ParamField>

<ParamField query="invoiceNumber" type="string">
  Filter by invoice number
</ParamField>

<ParamField query="customerId" type="integer">
  Filter by customer ID
</ParamField>

<ParamField query="customerName" type="string">
  Filter by customer name (partial match)
</ParamField>

<ParamField query="searchTerm" type="string">
  Search across invoice description and notes
</ParamField>

<ParamField query="fromIssueDate" type="string">
  Filter invoices issued on or after this date (ISO 8601)
</ParamField>

<ParamField query="toIssueDate" type="string">
  Filter invoices issued on or before this date (ISO 8601)
</ParamField>

<ParamField query="fromDueDate" type="string">
  Filter invoices due on or after this date (ISO 8601)
</ParamField>

<ParamField query="toDueDate" type="string">
  Filter invoices due on or before this date (ISO 8601)
</ParamField>

<ParamField query="minAmount" type="number">
  Filter by minimum total amount
</ParamField>

<ParamField query="maxAmount" type="number">
  Filter by maximum total amount
</ParamField>

<ParamField query="isOverdue" type="boolean">
  Filter only overdue invoices
</ParamField>

<ParamField query="isPaid" type="boolean">
  Filter only paid invoices
</ParamField>

<ParamField query="isRecurring" type="boolean">
  Filter only recurring invoices
</ParamField>

<ParamField query="sortBy" type="string">
  Sort results by field. Supported values: `invoiceNumber`, `customerName`, `dueDate`, `total`, `status`, `createdAt`
</ParamField>

<ParamField query="sortDirection" type="string" default="desc">
  Sort direction: `asc` or `desc`
</ParamField>

## Response

Returns a paginated list of invoice objects.

<ResponseField name="data" type="array">
  Array of invoice objects
</ResponseField>

<ResponseField name="total" type="integer">
  Total number of invoices matching the filters
</ResponseField>

<ResponseField name="page" type="integer">
  Current page number
</ResponseField>

<ResponseField name="pageSize" type="integer">
  Number of items per page
</ResponseField>

```bash cURL

  --header 'Content-Type: application/json'
```

});
