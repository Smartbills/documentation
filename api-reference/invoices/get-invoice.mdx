---
title: "Get Invoice"
api: "GET https://api.smartbills.io/v1/businesses/{businessId}/invoices/{invoiceId}"
description: "Retrieve a specific invoice by ID"
---

## Path Parameters

<ParamField path="businessId" type="integer" required>
  The unique identifier of the business
</ParamField>

<ParamField path="invoiceId" type="integer" required>
  The unique identifier of the invoice
</ParamField>

## Response

<ResponseField name="id" type="integer">
  Unique invoice identifier
</ResponseField>

<ResponseField name="invoiceNumber" type="string">
  Human-readable invoice number (e.g., INV-2026-0001)
</ResponseField>

<ResponseField name="customerId" type="integer">
  ID of the customer this invoice is billed to
</ResponseField>

<ResponseField name="totalAmount" type="number">
  Total invoice amount
</ResponseField>

<ResponseField name="currency" type="string">
  ISO 4217 currency code (e.g., CAD, USD)
</ResponseField>

<ResponseField name="billStatus" type="string">
  Invoice status: `Draft`, `Sent`, `Paid`, `Overdue`, `Void`, `Cancelled`
</ResponseField>

<ResponseField name="dueDate" type="string">
  Payment due date in ISO 8601 format
</ResponseField>

<ResponseField name="isOverdue" type="boolean">
  Whether the invoice is past its due date
</ResponseField>

<ResponseField name="isPaid" type="boolean">
  Whether the invoice has been fully paid
</ResponseField>

<ResponseField name="paymentLinkToken" type="string">
  Unique token for the hosted payment page
</ResponseField>

<ResponseField name="paymentLinkUrl" type="string">
  URL for the hosted payment page
</ResponseField>

<ResponseField name="stripePaymentIntentId" type="string">
  Stripe PaymentIntent ID if payment was processed via Stripe
</ResponseField>

<ResponseField name="sentAt" type="string">
  Timestamp when the invoice was sent to the customer
</ResponseField>

<ResponseField name="viewedAt" type="string">
  Timestamp when the customer first viewed the invoice
</ResponseField>

<ResponseField name="voidedAt" type="string">
  Timestamp when the invoice was voided
</ResponseField>

<ResponseField name="paidOn" type="string">
  Timestamp when the invoice was paid
</ResponseField>

<ResponseField name="amountDue" type="object">
  <Expandable title="properties">
    <ResponseField name="amount" type="number">Amount due</ResponseField>
    <ResponseField name="currency" type="string">Currency code</ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="lineItems" type="array">
  <Expandable title="properties">
    <ResponseField name="id" type="integer">Line item ID</ResponseField>
    <ResponseField name="description" type="string">Item description</ResponseField>
    <ResponseField name="quantity" type="number">Quantity</ResponseField>
    <ResponseField name="unitPrice" type="number">Price per unit</ResponseField>
    <ResponseField name="totalAmount" type="number">Line total</ResponseField>
    <ResponseField name="productId" type="integer">Associated product ID</ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="createdAt" type="string">
  Creation timestamp
</ResponseField>

<ResponseField name="updatedAt" type="string">
  Last update timestamp
</ResponseField>

<RequestExample>

```bash cURL
curl --request GET \
  --url https://api.smartbills.io/v1/businesses/456/invoices/1001 \
  --header 'Authorization: Bearer YOUR_API_KEY' \
  --header 'Content-Type: application/json'
```

```javascript JavaScript
const invoice = await smartbills.invoices.getById(1001);
```

```csharp C#
var invoice = await invoiceClient.GetByIdAsync(1001,
    new SBRequestOptions { BusinessId = 456 });
```

</RequestExample>
