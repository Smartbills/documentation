title: "List Departments"

api: "GET https://api.smartbills.io/v1/businesses/{businessId}/departments"
---

## Overview

Retrieve a list of departments configured for a business. Departments are used to associate expenses with organizational units for cost tracking and reporting.

## Authentication

<ParamField header="Authorization" type="string" required>
  Bearer token. Format: `Bearer YOUR_API_KEY`
</ParamField>

<ParamField header="x-tenant-id" type="string" required>
  Business ID for multi-tenant context
</ParamField>

## Path Parameters

<ParamField path="businessId" type="integer" required>
  The unique identifier of the business
</ParamField>

## Query Parameters

<ParamField query="page" type="integer" default="1">
  Page number for pagination (starts at 1)
</ParamField>

<ParamField query="pageSize" type="integer" default="20">
  Number of items per page (max: 100)
</ParamField>

<ParamField query="search" type="string">
  Search departments by name or code (partial match)
</ParamField>

<ParamField query="enabled" type="boolean">
  Filter by enabled status. `true` returns only active departments, `false` returns only disabled departments. Omit to return all.
</ParamField>

## Response

<ResponseField name="data" type="array">
  Array of department objects

  <Expandable title="Department Object">
    <ResponseField name="id" type="integer">
      Unique department identifier
    </ResponseField>

    <ResponseField name="name" type="string">
      Department name
    </ResponseField>

    <ResponseField name="code" type="string">
      Department code for internal reference
    </ResponseField>

    <ResponseField name="costCenter" type="string">
      Cost center identifier for accounting
    </ResponseField>

    <ResponseField name="defaultApproverId" type="integer">
      ID of the default approver for this department
    </ResponseField>

    <ResponseField name="enabled" type="boolean">
      Whether the department is active
    </ResponseField>

    <ResponseField name="createdAt" type="string">
      Timestamp when the department was created (ISO 8601)
    </ResponseField>

    <ResponseField name="updatedAt" type="string">
      Timestamp of last update (ISO 8601)
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="pagination" type="object">
  Pagination metadata

  <Expandable title="Pagination Object">
    <ResponseField name="page" type="integer">
      Current page number
    </ResponseField>

    <ResponseField name="pageSize" type="integer">
      Number of items per page
    </ResponseField>

    <ResponseField name="totalPages" type="integer">
      Total number of pages
    </ResponseField>

    <ResponseField name="totalCount" type="integer">
      Total number of departments matching the filters
    </ResponseField>

    <ResponseField name="hasNext" type="boolean">
      Whether there is a next page
    </ResponseField>

    <ResponseField name="hasPrevious" type="boolean">
      Whether there is a previous page
    </ResponseField>
  </Expandable>
</ResponseField>

<RequestExample>

```bash cURL
curl --request GET \
  --url 'https://api.smartbills.io/v1/businesses/456/departments?page=1&pageSize=20&enabled=true' \
  --header 'Authorization: Bearer YOUR_API_KEY' \
  --header 'x-tenant-id: 456'
```

```javascript JavaScript SDK
import { SBClient } from "@smartbills/sdk";

const client = new SBClient({ apiKey: "YOUR_API_KEY" });

const departments = await client.departments.list({
  search: "Engineering",
  enabled: true,
  page: 1,
  pageSize: 20
});

console.log(`Found ${departments.pagination.totalCount} departments`);
```

```python Python SDK
from smartbills import Smartbills

client = Smartbills(api_key="YOUR_API_KEY")

departments = await client.departments.list()

for dept in departments.data:
    print(f"{dept.name} ({dept.code}) - Cost Center: {dept.cost_center}")
```
