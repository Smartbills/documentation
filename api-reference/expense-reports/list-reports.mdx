---
title: "List Expense Reports"
api: "GET https://api.smartbills.io/v1/businesses/{businessId}/expense-reports"
description: "Retrieve a list of expense reports for a business"
---

## Path Parameters

<ParamField path="businessId" type="integer" required>
  The unique identifier of the business
</ParamField>

## Query Parameters

<ParamField query="page" type="integer" default="1">
  Page number for pagination
</ParamField>

<ParamField query="pageSize" type="integer" default="20">
  Number of items per page (max: 100)
</ParamField>

<ParamField query="status" type="string">
  Filter by status: `draft`, `submitted`, `approved`, `partially_approved`, `rejected`, `reimbursed`
</ParamField>

<ParamField query="userId" type="integer">
  Filter by user ID (submitter)
</ParamField>

<ParamField query="startDate" type="string">
  Filter reports from this date
</ParamField>

<ParamField query="endDate" type="string">
  Filter reports until this date
</ParamField>

## Response

<ResponseField name="data" type="array">
  Array of expense report objects
</ResponseField>

<ResponseField name="pagination" type="object">
  Pagination information
</ResponseField>

import requests
