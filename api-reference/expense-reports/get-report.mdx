---
title: "Get Expense Report"
api: "GET https://api.smartbills.io/v1/businesses/{businessId}/expense-reports/{reportId}"
description: "Retrieve a specific expense report with full details including transactions"
---

## Overview

Retrieve detailed information about a specific expense report, including all associated transactions, approval history, and comments.

## Authentication

<ParamField header="Authorization" type="string" required>
  Bearer token. Format: `Bearer YOUR_API_KEY`
</ParamField>

<ParamField header="x-tenant-id" type="string" required>
  Business ID for multi-tenant context
</ParamField>

## Path Parameters

<ParamField path="businessId" type="integer" required>
  The unique identifier of the business
</ParamField>

<ParamField path="reportId" type="integer" required>
  The unique identifier of the expense report
</ParamField>

## Query Parameters

<ParamField query="sortBy" type="string" default="date">
  Field to sort transactions by. Options: `date`, `amount`, `merchant`, `createdAt`
</ParamField>

<ParamField query="sortOrder" type="string" default="desc">
  Sort order for transactions. Options: `asc`, `desc`
</ParamField>

## Response

<ResponseField name="id" type="integer">
  Unique expense report identifier
</ResponseField>

<ResponseField name="title" type="string">
  Report title
</ResponseField>

<ResponseField name="description" type="string">
  Report description
</ResponseField>

<ResponseField name="businessPurpose" type="string">
  Business purpose for the expenses
</ResponseField>

<ResponseField name="status" type="string">
  Report status: `draft`, `submitted`, `approved`, `partially_approved`, `rejected`, `changes_required`, `reimbursed`
</ResponseField>

<ResponseField name="totalAmount" type="number">
  Total amount of all transactions in the report
</ResponseField>

<ResponseField name="approvedAmount" type="number">
  Total approved amount (set after approval), null if not yet approved
</ResponseField>

<ResponseField name="reportCurrency" type="string">
  Three-letter ISO currency code (e.g., USD, CAD, EUR)
</ResponseField>

<ResponseField name="expenseCount" type="integer">
  Number of transactions in the report
</ResponseField>

<ResponseField name="employeeId" type="integer">
  ID of the employee who owns the report
</ResponseField>

<ResponseField name="startDate" type="string">
  Expense period start date (YYYY-MM-DD)
</ResponseField>

<ResponseField name="endDate" type="string">
  Expense period end date (YYYY-MM-DD)
</ResponseField>

<ResponseField name="transactions" type="array">
  Array of transaction objects included in the report

  <Expandable title="Transaction Object">
    <ResponseField name="id" type="integer">
      Transaction ID
    </ResponseField>

    <ResponseField name="amount" type="number">
      Transaction amount
    </ResponseField>

    <ResponseField name="currency" type="string">
      Transaction currency code
    </ResponseField>

    <ResponseField name="merchant" type="string">
      Merchant or vendor name
    </ResponseField>

    <ResponseField name="date" type="string">
      Transaction date (YYYY-MM-DD)
    </ResponseField>

    <ResponseField name="category" type="string">
      Expense category name
    </ResponseField>

    <ResponseField name="notes" type="string">
      Transaction notes
    </ResponseField>

    <ResponseField name="hasReceipt" type="boolean">
      Whether a receipt is attached
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="comments" type="array">
  Array of comment objects on the report

  <Expandable title="Comment Object">
    <ResponseField name="id" type="integer">
      Comment ID
    </ResponseField>

    <ResponseField name="userId" type="integer">
      ID of the user who posted the comment
    </ResponseField>

    <ResponseField name="userName" type="string">
      Name of the user who posted the comment
    </ResponseField>

    <ResponseField name="comment" type="string">
      Comment text
    </ResponseField>

    <ResponseField name="createdAt" type="string">
      Timestamp when the comment was posted (ISO 8601)
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="submittedAt" type="string">
  Timestamp when the report was submitted (ISO 8601), null if not submitted
</ResponseField>

<ResponseField name="approvedAt" type="string">
  Timestamp when the report was approved (ISO 8601), null if not approved
</ResponseField>

<ResponseField name="createdAt" type="string">
  Timestamp when the report was created (ISO 8601)
</ResponseField>

<ResponseField name="updatedAt" type="string">
  Timestamp of last update (ISO 8601)
</ResponseField>

<RequestExample>

```bash cURL
curl --request GET \
  --url 'https://api.smartbills.io/v1/businesses/456/expense-reports/1001?sortBy=date&sortOrder=desc' \
  --header 'Authorization: Bearer YOUR_API_KEY' \
  --header 'x-tenant-id: 456'
```

```javascript JavaScript SDK
import { SBClient } from "@smartbills/sdk";

const client = new SBClient({ apiKey: "YOUR_API_KEY" });

const report = await client.expenseReports.getById(1001, {
  sortBy: "date",
  sortOrder: "desc"
});

console.log("Report:", report.title);
console.log("Total:", report.totalAmount, report.reportCurrency);
```

```python Python SDK
from smartbills import Smartbills

client = Smartbills(api_key="YOUR_API_KEY")

report = client.expense_reports.get_by_id(1001)

print(f"Report: {report.title}")
print(f"Total: {report.total_amount} {report.report_currency}")
```

</RequestExample>

## Error Responses

| Status Code | Error Code | Description |
|------------|------------|-------------|
| 401 | UNAUTHORIZED | Missing or invalid API key |
| 403 | FORBIDDEN | Insufficient permissions to access this report |
| 404 | NOT_FOUND | Expense report not found |
| 404 | BUSINESS_NOT_FOUND | Business ID does not exist |
| 429 | RATE_LIMIT_EXCEEDED | Too many requests |
