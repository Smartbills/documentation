---
title: "Get Expense"
api: "GET https://api.smartbills.io/v1/expenses/{expenseId}"
description: "Retrieve a specific expense by ID"
---

## Path Parameters

<ParamField path="expenseId" type="integer" required>
  The unique identifier of the expense
</ParamField>

## Response

<ResponseField name="id" type="integer">
  Unique expense identifier
</ResponseField>

<ResponseField name="amount" type="number">
  Expense amount
</ResponseField>

<ResponseField name="currency" type="string">
  ISO 4217 currency code (e.g., USD, CAD, EUR)
</ResponseField>

<ResponseField name="date" type="string">
  Transaction date in ISO 8601 format
</ResponseField>

<ResponseField name="merchant" type="string">
  Merchant or vendor name
</ResponseField>

<ResponseField name="category" type="string">
  Expense category
</ResponseField>

<ResponseField name="status" type="string">
  Expense status: `pending`, `approved`, `rejected`, `reimbursed`
</ResponseField>

<ResponseField name="description" type="string">
  Expense description or notes
</ResponseField>

<ResponseField name="userId" type="integer">
  ID of the user who created the expense
</ResponseField>

<ResponseField name="businessId" type="integer">
  ID of the business this expense belongs to
</ResponseField>

<ResponseField name="attachmentUrl" type="string">
  URL of the receipt/invoice attachment
</ResponseField>

<ResponseField name="tags" type="array">
  Array of tags associated with the expense
</ResponseField>

<ResponseField name="metadata" type="object">
  Custom metadata object
</ResponseField>

<ResponseField name="createdAt" type="string">
  Creation timestamp in ISO 8601 format
</ResponseField>

<ResponseField name="updatedAt" type="string">
  Last update timestamp in ISO 8601 format
</ResponseField>

<RequestExample>

```bash cURL
curl --request GET \
  --url https://api.smartbills.io/v1/expenses/12345 \
  --header 'Authorization: Bearer YOUR_API_KEY' \
  --header 'Content-Type: application/json'
```

```javascript JavaScript
const response = await fetch(
  'https://api.smartbills.io/v1/expenses/12345',
  {
    method: 'GET',
    headers: {
      'Authorization': 'Bearer YOUR_API_KEY',
      'Content-Type': 'application/json'
    }
  }
);

const expense = await response.json();
console.log(expense);
```

```python Python
import requests

url = 'https://api.smartbills.io/v1/expenses/12345'
headers = {
    'Authorization': 'Bearer YOUR_API_KEY',
    'Content-Type': 'application/json'
}

response = requests.get(url, headers=headers)
expense = response.json()
print(expense)
```
