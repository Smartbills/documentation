---
title: "List Pending Approvals"
description: "Retrieve a list of expense reports pending your approval"
api: "GET https://api.smartbills.io/v1/businesses/{businessId}/approbations/pending"
---

## Overview

Retrieve all expense reports that are awaiting your approval. This endpoint returns reports submitted by employees who are in your approval chain. Results are sorted by submission date, with the most recent first.

## Authentication

<ParamField header="Authorization" type="string" required>
  Bearer token for authentication. Format: `Bearer YOUR_API_KEY`
</ParamField>

## Path Parameters

<ParamField path="businessId" type="integer" required>
  The unique identifier of the business
</ParamField>

## Query Parameters

<ParamField query="page" type="integer" default="1">
  Page number for pagination
</ParamField>

<ParamField query="pageSize" type="integer" default="20">
  Number of items per page (max: 100)
</ParamField>

<ParamField query="employeeId" type="integer">
  Filter by submitting employee ID
</ParamField>

<ParamField query="minAmount" type="number">
  Filter by minimum report amount
</ParamField>

<ParamField query="maxAmount" type="number">
  Filter by maximum report amount
</ParamField>

<ParamField query="sortBy" type="string" default="submittedAt">
  Sort by field. Options: `submittedAt`, `totalAmount`, `employeeName`
</ParamField>

<ParamField query="sortOrder" type="string" default="desc">
  Sort order. Options: `asc`, `desc`
</ParamField>

## Response

<ResponseField name="data" type="array">
  Array of pending approval objects

  <Expandable title="Pending Approval Object">
    <ResponseField name="reportId" type="integer">
      ID of the expense report
    </ResponseField>

    <ResponseField name="reportName" type="string">
      Name of the expense report
    </ResponseField>

    <ResponseField name="employeeId" type="integer">
      ID of the employee who submitted the report
    </ResponseField>

    <ResponseField name="employeeName" type="string">
      Full name of the submitting employee
    </ResponseField>

    <ResponseField name="totalAmount" type="number">
      Total amount of the report
    </ResponseField>

    <ResponseField name="currency" type="string">
      ISO 4217 currency code
    </ResponseField>

    <ResponseField name="expenseCount" type="integer">
      Number of expenses in the report
    </ResponseField>

    <ResponseField name="submittedAt" type="string">
      Timestamp when the report was submitted (ISO 8601)
    </ResponseField>

    <ResponseField name="description" type="string">
      Report description
    </ResponseField>

    <ResponseField name="approvalLevel" type="integer">
      Current approval level in the chain (1 = first approver, 2 = second, etc.)
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="metadata" type="object">
  Pagination metadata

  <Expandable title="Metadata Object">
    <ResponseField name="page" type="integer">
      Current page number
    </ResponseField>
    <ResponseField name="pageSize" type="integer">
      Items per page
    </ResponseField>
    <ResponseField name="totalPages" type="integer">
      Total number of pages
    </ResponseField>
    <ResponseField name="count" type="integer">
      Total number of pending approvals
    </ResponseField>
  </Expandable>
</ResponseField>
